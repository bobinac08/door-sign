    // 1. DYNAMIC CLOCK & DATE

    function updateClock() {

      const now = new Date();

      

      // Time formatting (e.g., 9:41 AM)

      const timeOptions = { hour: 'numeric', minute: '2-digit', hour12: true };

      document.getElementById('time').innerText = now.toLocaleTimeString([], timeOptions);


      // Date formatting (e.g., Mon, Jul 27)

      const dateOptions = { weekday: 'short', month: 'short', day: 'numeric' };

      document.getElementById('date').innerText = now.toLocaleDateString([], dateOptions);

    }

    setInterval(updateClock, 1000);

    updateClock();


    // 2. AUTOMATIC LOCATION BY WORK SCHEDULE

    function updateLocationBySchedule() {

      const now = new Date();

      const day = now.getDay(); // 0 = Sun, 1 = Mon, ..., 6 = Sat

      const hour = now.getHours();

      

      const badge = document.getElementById('location-badge');


      // Monday to Friday (1-5)

      if (day >= 1 && day <= 5) {

        if (hour >= 8 && hour < 17) { // 8 AM to 5 PM

          badge.innerText = "In Office";

          badge.style.color = "#4ade80"; // Green

        } else {

          badge.innerText = "Off the Clock";

          badge.style.color = "#94a3b8"; // Muted gray

        }

      } else {

        badge.innerText = "Out of Office";

        badge.style.color = "#f87171"; // Soft red

      }

    }

    setInterval(updateLocationBySchedule, 60000); // Check every minute

    updateLocationBySchedule();


    // 3. AUTO-REFRESH OUTLOOK IFRAME EVERY 10 MINUTES

    setInterval(() => {

      const outlookIframe = document.getElementById('outlook-iframe');

      const currentSrc = outlookIframe.src.split('?')[0];

      // Appends a timestamp to bypass browser cache and fetch fresh calendar data

      outlookIframe.src = currentSrc + '?refresh=' + new Date().getTime();

    }, 600000); // 600,000 ms = 10 mins
